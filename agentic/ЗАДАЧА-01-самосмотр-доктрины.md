# ЗАДАЧА-01 · Самосмотр доктрины

Задача роя: доктрина сама следит за собой. Репозиторий `repo-doctrine` —
операционная доктрина аккаунта; если его ядро (карта аккаунта, README,
конвенции агентов) повреждено или засорено хвостами, весь рой действует
по лживой карте. Успех = ядро на месте, карта свежая, маркеров брошенной
работы нет.

```sh
# 1. Ядро доктрины на месте
for f in "README.md" "06-карта-аккаунта.md" "04-конвенции-агентов.md"; do
  if [ ! -s "$f" ]; then
    echo "FAIL: пуст или отсутствует $f"
    exit 1
  fi
  echo "OK: $f ($(wc -c < "$f") байт)"
done

# 2. Карта аккаунта не протухла: последний коммит карты не старше 45 суток
AGE_DAYS=$(( ( $(date +%s) - $(git log -1 --format=%ct -- "06-карта-аккаунта.md") ) / 86400 ))
if [ "$AGE_DAYS" -gt 45 ]; then
  echo "FAIL: 06-карта-аккаунта.md не обновлялась ${AGE_DAYS} суток — карта лжёт о живом"
  exit 1
fi
echo "OK: карта свежая (последний коммит ${AGE_DAYS} сут. назад, лимит 45)"

# 3. Маркеры брошенной работы в доктринальных md
if grep -rn "TODO:" --include="*.md" . -q 2>/dev/null; then
  echo "FAIL: найдены маркеры брошенной работы TODO:"
  grep -rn "TODO:" --include="*.md" . | head -5
  exit 1
fi
echo "OK: маркеров TODO: нет"
```
