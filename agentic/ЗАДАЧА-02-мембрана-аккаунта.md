# ЗАДАЧА-02 · Мембрана аккаунта

Задача роя: мембрана (CODEOWNERS, SECURITY.md, AGENTS.md, LICENSE) обязана
стоять на каждой публичной клетке роя — так чужой код не проходит к ядру
мимо владельца, а уязвимости имеют приватный канал. Ростер клеток —
`agentic/РОСТЕР-АККАУНТА.json` (зеркало канона `edinyy-svod/sbom/…`, T01;
сам SBOM приватен, а GITHUB_TOKEN репо-скоупа даёт 401 на чужие репо и
анонимный REST ограничен 60/ч — уроки живых прогонов ×8, поэтому транспорт
клеток = git-протокол: `git ls-tree` из безблочных клонов, ноль REST API).
Успех = у каждой публичной живой клетки все 4 файла мембраны на месте
(в любой канонической локации).

```sh
python3 - <<'PYEOF'
import json, os, subprocess

roster = json.load(open('agentic/РОСТЕР-АККАУНТА.json', encoding='utf-8'))
cells = roster['cells']
print(f"ростер (снимок {roster['snapshot'][:10]}): {len(cells)} клеток")

# канонические локации мембраны (docs.github.com): корень, .github/, docs/
# сравнение в верхнем регистре с ОБЕИХ сторон (пути upper-сятся)
CANON = {
    'CODEOWNERS': {'CODEOWNERS', '.GITHUB/CODEOWNERS', 'DOCS/CODEOWNERS'},
    'SECURITY.MD': {'SECURITY.MD', '.GITHUB/SECURITY.MD', 'DOCS/SECURITY.MD'},
    'AGENTS.MD': {'AGENTS.MD', '.GITHUB/AGENTS.MD'},
    'LICENSE': {'LICENSE', 'LICENSE.MD', 'LICENSE.TXT', '.GITHUB/LICENSE'},
}

fail, ok, skip = [], 0, 0
for cell in cells:
    name = cell['name']
    if cell['archived'] == 'true' or cell['visibility'] == 'private':
        skip += 1
        continue
    dest = f'/tmp/membrane/{name}'
    subprocess.run(['rm', '-rf', dest], check=True)
    rc = subprocess.run(
        ['git', 'clone', '-q', '--depth', '1', '--filter=blob:none',
         f'https://github.com/rtynbirf/{name}.git', dest],
        capture_output=True, text=True, timeout=180)
    if rc.returncode != 0:
        fail.append(f'{name}: клон не удался ({(rc.stderr or "").strip()[:70]})')
        continue
    r = subprocess.run(['git', 'ls-tree', '-r', 'HEAD', '--name-only'],
                       cwd=dest, capture_output=True, text=True, timeout=120)
    paths = set(p.upper() for p in r.stdout.splitlines())
    missing = [f for f, locs in CANON.items() if not (paths & locs)]
    if missing:
        fail.append(f'{name}: нет {", ".join(missing)}')
    else:
        ok += 1

print(f'мембрана: OK {ok} · skip {skip} (private/archived) · FAIL {len(fail)}')
for f in fail:
    print('  ❌', f)
assert not fail, 'мембрана дырявая: см. список выше'
print('OK: мембрана публичных клеток целая')
PYEOF
```
