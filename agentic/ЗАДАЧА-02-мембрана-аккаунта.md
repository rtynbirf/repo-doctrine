# ЗАДАЧА-02 · Мембрана аккаунта

Задача роя: мембрана (CODEOWNERS, SECURITY.md, AGENTS.md, LICENSE) обязана
стоять на каждой публичной клетке роя — так чужой код не проходит к ядру
мимо владельца, а уязвимости имеют приватный канал. Ростер клеток берётся
из аккаунт-SBOM роя (`edinyy-svod`, T01) — единый источник правды.
Успех = у каждой публичной клетки все 4 файла мембраны на месте.

```sh
python3 - <<'PYEOF'
import json, os, urllib.request, urllib.error

API = 'https://api.github.com'
tok = os.environ.get('GITHUB_TOKEN', '')
h = {'User-Agent': 'agentic-pilot-membrane'}
if tok:
    h['Authorization'] = f'Bearer {tok}'

def get(url):
    req = urllib.request.Request(url, headers=dict(h, Accept='application/vnd.github+json'))
    try:
        with urllib.request.urlopen(req, timeout=30) as r:
            return r.status
    except urllib.error.HTTPError as e:
        return e.code

def raw(url):
    req = urllib.request.Request(url, headers=dict(h, Accept='application/vnd.github.raw'))
    with urllib.request.urlopen(req, timeout=40) as r:
        return r.read()

# ростер живёт в SBOM роя (edinyy-svod, main) — единый источник правды
bom = json.loads(raw(f'{API}/repos/rtynbirf/edinyy-svod/contents/sbom/account-sbom.cdx.json'))
comps = bom['components']
print(f'ростер из SBOM: {len(comps)} клеток')

fail, ok, skip = [], 0, 0
# канонические локации мембраны (docs.github.com): корень, .github/, docs/
LOCATIONS = {
    'CODEOWNERS': ['CODEOWNERS', '.github/CODEOWNERS', 'docs/CODEOWNERS'],
    'SECURITY.md': ['SECURITY.md', '.github/SECURITY.md', 'docs/SECURITY.md'],
    'AGENTS.md': ['AGENTS.md', '.github/AGENTS.md'],
    'LICENSE': ['LICENSE', 'LICENSE.md', 'LICENSE.txt', '.github/LICENSE'],
}
for c in comps:
    props = {p['name']: p['value'] for p in c.get('properties', [])}
    name = c['name']
    if props.get('rtynbirf:archived') == 'true' or props.get('rtynbirf:visibility') == 'private':
        skip += 1
        continue
    missing = []
    for f, paths in LOCATIONS.items():
        if not any(get(f'https://api.github.com/repos/rtynbirf/{name}/contents/{p}') == 200 for p in paths):
            missing.append(f)
    if missing:
        fail.append(f'{name}: нет {", ".join(missing)}')
    else:
        ok += 1

print(f'мембрана: OK {ok} · skip {skip} (private/archived) · FAIL {len(fail)}')
for f in fail:
    print('  ❌', f)
assert not fail, 'мембрана дырявая: см. список выше'
print('OK: мембрана публичных клеток целая')
PYEOF
```
